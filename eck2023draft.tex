We now motivate the structure of the Full House Model. Let $N_i$ denote the size of population $i$. We will suppose the every individual $k = 1,\ldots, N_i$ has an underlying latent aptitude 
$X_{i,1}, \ldots, X_{i,N_i} \stackrel{\text { iid }}{\sim} F_{X}$. 
We denote $X_{i,(j)}$ as the $j$th ordered latent aptitude in population $i$.
We define $g_{i}\left(\cdot, \cdot\right)$ as the system inclusion function, where 
$g_{i}\left(X_{i,j}, \textbf{X}_{i,-j}\right)=1$ indicates that individual $j$ is included in the $i$th system, 
and $g_{i}\left(X_{i,j}, \textbf{X}_{i,-j}\right)=0$ indicates that individual $j$ is not included in the $i$th system. Let $\textbf{X}_{i,-j}$ be the vector of all latent aptitudes not including the component $j$.

We now suppose that an individual $j$ included in system $i$ has an observed statistic $Y_{i,j}$ where  
we suppose that $Y_{i,1}, \ldots, Y_{i,n_i} \overset{iid}{\sim} F_{Y_i}$, $F_{Y_i}$ is continuous, and $n_i < N_i$ is the number of individuals included in system $i$. We denote $Y_{i,(j)}$ as the $j$th ordered statistic for system $i$.

The key to the Full House Model is connecting latent aptitudes $X$ with observed statistics $Y$. 
To achieve this we assume that we can form pairs $(Y_{i,(j)}, X_{i,(r(N_i, n_i, j))})$, 
$j = 1,\ldots,n_i$, where $r(N_i, n_i, k)$ is the index function that connects this latent aptitude to the $k$th components in the system $i$. The function $r(N_i, n_i, k)$ determines the relationship between the components in the system and the selected latent aptitudes in the population. As an example, $r(N_i, n_i, k) = N_i - n_i + k$, then the highest observed statistics $Y_{i,j}$ in system i will be assumed to have the highest latent aptitudes in population $i$.

We now demonstrate how the Full House Model works in a parametric setting. Consider the pair 
$(Y_{i,(j)}, X_{i,(r(N_i, n_i, j))})$ 
%(\textcolor{red}{Needs to be more general, this currently corresponds to the case where all the highest latent aptitudes are in the system.})
and suppose that the distribution corresponding to $Y_{i, j}$ from the $i$th system is known to belong to a continuous parametric family indexed by unknown parameter $\theta_{i}$, and let $F_{Y_i}(\cdot \mid \theta_i)$ be a parametric CDF with parameter $\theta_i \in \mathbb{R}^{p_{Y_i}}$. We can estimate $\theta_i$ with $\hat{\theta_i}$ and plug the estimator into the CDF $F_{Y}(\cdot \mid \hat{\theta_i})$.

In order to connect the  $Y_{i,(j)}$ and $X_{i,(r(N_i, n_i, j))}$ and obtain an estimate of the underlying talents,  we will make use of the following classical order statistics properties, 
$$
\begin{aligned}
F_{Y_i}\left(Y_{i,(j)} \mid \theta_i\right) & \sim U_{i,(j)}, & & F_{Y_i}\left(Y_{i,(j)} \mid \hat{\theta_i}\right) \approx U_{i,(j)}, \\
F_{Y_{i,(j)}}\left(Y_{i,(j)} \mid \theta_i\right) & \sim U_{i,j}, & & F_{Y_{i,(j)}}\left(Y_{i,(j)} \mid \hat{\theta_i}\right) \approx U_{i,j},
\end{aligned}
$$
where $\approx$ means approximately distributed, $\sim$ means distributed as, $U_{i,j} \sim U(0,1)$, and $U_{i,(j)} \sim \operatorname{Beta}(j, n_i+1-j)$ and the quality of the approximation in the right hand side depends upon the estimator $\hat{\theta}_i$ and the sample size. 

We now connect the order statistics to the underlying talent distribution that comes from a population with $N_i \geq n_i$ observations when $F_{X}$ is known. This connection is established with 
\begin{equation} \label{eq:trueparX}
F_{X_{i, (r(N_i, n_i, j))}}^{-1}\left(F_{Y_{i,(j)}}\left(Y_{i,(j)} \mid \theta_i\right)\right) 
  \sim F_{X_{i, (r(N_i, n_i, j))}}^{-1}\left(U_{i,j}\right) 
  = X_{i, (r(N_i, n_i, j))}.
\end{equation}

\section*{Paradoxes}


Consider two populations $i = 1,2$ with sizes $N_1$ and $N_2$, and systems of sizes $n_1$ and $n_2$. We define a paradox (or a Gould paradox) when 
$$
  X_{1,(N_1)} > X_{2,(N_2)} \quad \text{and} \quad Y_{1,(n_1)} < Y_{2,(n_2)}
$$
both occur. A central interest is evaluating the probability of such a paradox, 
\begin{equation} \label{eq:simpleparadox}
  \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right).
\end{equation}
As a simple illustrative example of some mechanics concerning paradoxes, suppose that $Y_{1,j} < Y_{2,k}$ a.s. Then \eqref{eq:simpleparadox} is equivalent to $\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}\right)$, and we have $\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}\right) \to 1$ as $N_1 \to \infty$.





Consider the setting where $g_{i}\left(X_{i, j}, \mathbf{X}_{i,-j}\right) = 1\left(X_{i,j} \geq X_{i,(N_i-n_i + 1)}\right)$. Then \eqref{eq:trueparY} gives 
$$
  F_{Y_{i, (n_i)}}^{-1}\left(F_X\left(X_{i, (N_i)}\right) | \theta_i\right) = Y_{i, (n_i)} \quad \text{for} \; i = 1,2.
$$
Let $f_i(\fatterdot) = F_{Y_{i, (n_i)}}^{-1}\left(F_X\left(\fatterdot\right) | \theta_i\right)$ and let $g_i(\fatterdot) = f_i^{-1}(\fatterdot)$.
%\begin{align*}
%  f_i(\fatterdot) &= F_{Y_{i, (n_i)}}^{-1}\left( F_{V(i,n_i)}\left(F_X\left(\fatterdot\right)\right) | \theta_i\right),	\\
%  g_i(\fatterdot) &= F_X^{-1}\left(F_{V(i,n_i)}^{-1}\left(F_{Y_{i,(n_i)}}\left(\fatterdot|\theta_i\right) \right) \right),
%\end{align*}
This construction yields,
\begin{align*} 
  \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
  = \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, f_1\left(X_{1,(N_1)}\right) < f_2\left(X_{2,(N_2)}\right)\right).
\end{align*}
Let $h_{i,k}(\fatterdot) = (g_i \circ f_k)(\fatterdot)$ when such an operation is defined. Then, 
\begin{align*} 
  &\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
  = \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, X_{1,(N_1)} < h_{1,2}\left(X_{2,(N_2)}\right)\right) \\
  &\qquad= \Prob\left( X_{2,(N_2)} < X_{1,(N_1)}  < h_{1,2}\left(X_{2,(N_2)}\right)\right).
\end{align*}



\begin{lemma} \label{lem:samedist}
If we assume 
\begin{align*}
  Y_{i,j} &\sim F_Y(y), \qquad i = 1,2, \qquad j = 1,\ldots,n_1, \\
  X_{i,j} &\sim F_X(x), \qquad i = 1,2, \qquad j = 1,\ldots,N_i, \\
\end{align*}
where $F_Y^{-1}(y)$ and $F_X^{-1}(x)$ are assumed to exist. Then,
\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
  = \frac{n_2}{n_1 + n_2} - \frac{N_2}{N_1 + N_2}.
\end{align*}
\end{lemma}

\begin{proof}
	\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) = \Prob\left( X_{2,(N_2)} < X_{1,(N_1)}  < h_{1,2}\left(X_{2,(N_2)}\right)\right) \\
	&= \int_{-\infty}^{+\infty}\left[\int_{x}^{h_{1,2}(x)} f_{X_{1,(N_1)}}(y)\ dy\right] \ f_{X_{2,(N_2)}}(x) \ dx \\
	&= \int_{-\infty}^{+\infty}\left[\int_{x}^{h_{1,2}(x)} N_1 (F_X(y))^{N_1-1}\ f_X(y)dy\right] \ N_2 (F_X(x))^{N_2-1}\ f_X(x) \ dx \\
	&= \int_{-\infty}^{+\infty}\left[\int_{x}^{F_X^{-1}(F_X(x)^{\frac{N_2}{N_1}\frac{n_1}{n_2}})} N_1 (F_X(y))^{N_1-1}\ f_X(y)dy\right] \ N_2 (F_X(x))^{N_2-1}\ f_X(x) \ dx \\
	&= \int_{-\infty}^{+\infty} (F_X(x)^{N_2\frac{n_1}{n_2}} - F_X(x)^{N_1}) \ N_2 (F_X(x))^{N_2-1}\ f_X(x) \ dx \\
	&= \int_0^1 t^{\frac{n_1}{n_2}} - t^{\frac{N_1}{N_2}} \ dt \\
	&= \frac{n_2}{n_1 + n_2} - \frac{N_2}{N_1 + N_2},
\end{align*}
where 
\begin{align*}
	&h_{1,2}(x) = (g_1 \circ f_2)(x) \\
	&= F_X^{-1}((F_Y(F_Y^{-1}(F_X^{\frac{N_2}{n_2}})))^{\frac{n_1}{N_1}})(x)\\
	&= F_X^{-1}(F_X^{\frac{N_2}{N_1}\frac{n_1}{n_2}})(x).
\end{align*}

\end{proof}

It is interesting to note that the probability of paradox $\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) = \frac{n_2}{n_1 + n_2}$ as $N_1 \to \infty$.

\subsection*{Example: Uniform distributions}

We will assume
\begin{align*}
  Y_{1,j} &\sim U(0,1), \qquad j = 1,\ldots,n_1, \\
  Y_{2,j} &\sim U(0,1+\delta), \qquad j = 1,\ldots,n_2, \\
  X_{i,j} &\sim U(0,1), \qquad j = 1,\ldots,N_i, \\
\end{align*}
where $\delta > 0$.

Based on the results above, we have

\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) = \Prob\left( X_{2,(N_2)} < X_{1,(N_1)}  < h_{1,2}\left(X_{2,(N_2)}\right)\right) \\
	&= \int_{0}^{1}\left[\int_{x}^{h_{1,2}(x)} f_{X_{1,(N_1)}}(y)\ dy\right] \ f_{X_{2,(N_2)}}(x) \ dx \\
	&= \int_{0}^{1}\left[\int_{x}^{h_{1,2}(x)} N_1 y^{N_1-1}\ dy\right] \ N_2 x^{N_2-1}\ dx \\
	&= \int_{0}^{1}\left[\int_{x}^{((1+\delta) x^{\frac{N_2}{n_2}})^\frac{n_1}{N_1}} N_1 y^{N_1-1}\ dy\right] \ N_2 x^{N_2-1} \ dx \\
	&= \int_{0}^{1} (((1+\delta)x^{\frac{N_2}{n_2}})^{n_1} - x^{N_1}) \ N_2 x^{N_2-1}\ dx \\
	& = \frac{n_2}{n_1 + n_2}(1+\delta)^{n_1} - \frac{N_2}{N_1 + N_2}.
	\end{align*}
Here we see that the probability of paradox is $1$ for all $\delta \geq [2(n_1/n_2 + 1)]^{1/n_1} - 1$. When $n_1 \gg n_2$, then $[2(n_1/n_2 + 1)]^{1/n_1} - 1 \approx n_1^{1/n_1} - 1$ and we see that the probability of paradox is $1$ for extremely small values of $\delta$. 



\subsection*{Stochastic dominance}


\begin{defn}
We will say that $Y_2$ is first order $\varepsilon$-weakly stochastically dominant over $Y_1$ if, for $\epsilon > 0$, 
$
  \Prob(Y_2 \leq y) \leq \Prob(Y_1 \leq y) \leq \Prob(Y_2 \leq y) + \varepsilon
$
for all $y$ and 
$
  \Prob(Y_2 \leq y) < \Prob(Y_1 \leq y) \geq \Prob(Y_2 \leq y) + \varepsilon
$
for some $y$.
\end{defn}


\begin{lemma}
Suppose that the system inclusion mechanism is $g(X_{i,j}) = 1(X_{i,j} > X_{i, N_i - n_i + 1})$ where 
\begin{align*}
  Y_{i,j} &\sim F_{Y_i}(y), \qquad i = 1,2, \qquad j = 1,\ldots,n_1, \\
  X_{i,j} &\sim F_X(x), \qquad i = 1,2, \qquad j = 1,\ldots,N_i, \\
\end{align*}
and $F_{Y_i}^{-1}(y)$ and $F_X^{-1}(x)$ exist. Further suppose that $Y_2$ is first order $\varepsilon$-weakly stochastically dominant over $Y_1$ for some $\epsilon > 0$. Then,
\begin{equation} \label{eq:paradox2}
  \frac{n_2}{n_1 + n_2} - \frac{N_2}{N_1 + N_2} \leq
  \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right).
\end{equation}
\end{lemma}

\begin{proof}
First, observe that the assumption of first order $\varepsilon$-weakly stochastically dominant implies that 
$$
  \left(F_{Y_{2,(n_2)}}(y)\right)^{n_1/n_2} \leq \left(F_{Y_1}(y)\right)^{n_1} \leq \left(F_{Y_2}(y) + \varepsilon\right)^{n_1} \leq \left(\left(F_{Y_{2,(n_2)}}(y)\right)^{1/n_2} + \varepsilon\right)^{n_1},
$$
so that 
$$
   F_X^{-1}\left(F_X^{\frac{N_2}{N_1}\frac{n_1}{n_2}}(X_{2,(N_2)})\right) \leq 
   h_{1,2}(X_{2,(N_2)}) \leq 
   F_{X_{1,(N_1)}}^{-1}\left(  \left(\left(F_{X_{2,(N_2)}}(X_{2,(N_2)})\right)^{1/n_2} + \varepsilon\right)^{n_1}  \right).
$$
\end{proof}
In combination with Lemma~\ref{lem:samedist} and the proof of Lemma~\ref{lem:samedist}, the first inequality implies that 
$$
  \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
  \geq \frac{n_2}{n_1 + n_2} - \frac{N_2}{N_1 + N_2}.
$$
This verifies the first inequality of \eqref{eq:paradox2}. We now verify the second inequality of \eqref{eq:paradox2}.
