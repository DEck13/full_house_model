\begin{prop}
	Define $g_{n,i}(x) = I_x(i, n-i+1)$, we want to prove $g^{-1}_{n, n-j}(x)^n$ is an increasing function with respect to n, or $g_{n, n-j}(x^{\frac{1}{n}})$ is a decreasing function with respect to n. We have known $g_{n,n-j}(x)$ is a decreasing function with respect to n and $g^{-1}_{n,n-j}(x)$ is an increasing function with respect to n. 
\end{prop}

I'm going to drop the $g_{i,n}$ notation here because it is very confusing. That notation should only be used when there are space constraints.

\vspace{12pt}

We will show that 
$$
  I_{\xn}(n-j, j+1) - I_{\xnone}(n-j+1, j+1) > 0.
$$

\vspace{12pt}

\noindent\textbf{Assume $\xn > \frac{n-j}{n}$}:

\vspace{12pt}

Integration by parts allows us to write
\begin{align*}
&I_{\xn}(n-j, j+1) - I_{\xnone}(n-j+1, j+1) \\ 
  &\qquad= I_{\xn}(n-j+1, j+1) - I_{\xnone}(n-j+1, j+1) + \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)} \\
  &\qquad= -\int_{\xn}^{\xnone} \frac{y^{n-j}(1 - y)^{j}}{B(n-j+1,j+1)} dy 
  	+ \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)}.
\end{align*}

The function $y^a(1-y)^b$ achieves its maximum value at $y^* = a/(a+b)$. The assumption that $\xn > \frac{n-j}{n}$ implies that the function $y^{n-j}(1 - y)^{j}$ is decreasing over $(\xn,\xnone)$. This implies that 
\begin{align*}
&-\int_{\xn}^{\xnone} \frac{y^{n-j}(1 - y)^{j}}{B(n-j+1,j+1)} dy 
  	+ \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)} \\
	&\qquad\geq \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j}(\xn  - \xnone)}{B(n-j+1,j+1)} + \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)}. 
\end{align*}
And $B(n-j+1,j+1) = B(n-j,j+1)\frac{n-j}{n+1}$ allows us to express the right hand side of the above inequality as
\begin{align*}
	&\frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j}(\xn  - \xnone)}{B(n-j+1,j+1)} + \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)} \\
	&\qquad= \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j}(\xn  - \xnone)}{B(n-j,j+1)\frac{n-j}{n+1}} + \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j+1}}{(n-j)B(n-j,j+1)} \\
	&\qquad= \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j}}{B(n-j,j+1)(n-j)}\left((n+1)\left(\xn - \xnone\right) + \left(1 - \xn\right)\right) \\
		&\qquad= \frac{\left(\xn\right)^{n-j}\left(1-\xn\right)^{j}}{(n-j)B(n-j,j+1)}\left(n\xn - (n+1)\xnone + 1\right)
\end{align*}
Let $f_{n}(x) = n\xn - (n+1)\xnone + 1$. Then 
\begin{align*}
	f^{'}_{n}(x) &= x^{\frac{1-n}{n}} - x^{-\frac{n}{n+1}} = x^{\frac{1-n}{n}}(1 - x^{-\frac{1}{n(n+1)}}).
\end{align*}
Thus, $f^{'}_{n}(x) \leq 0$ for all $x \in (0,1)$ which implies that $f_n(x) \geq 0$ for all $x \in (0,1)$. Therefore, when $\xn > \frac{n-j}{n}$ we can conclude that
$$
  I_{\xn}(n-j, j+1) - I_{\xnone}(n-j+1, j+1) \geq 0.
$$

\vspace{12pt}

\noindent\textbf{Assume $\xnone < \frac{n-j}{n}$}:

\vspace{12pt}






Now,
\begin{align*}
	f_{n,j}^{\prime}(x) &= \frac{(n+1)(n-j)}{j+1}\left(\frac{x^{\frac{1-n}{n}}}{n} - \frac{x^{-\frac{n}{n+1}}}{n+1}\right) - \frac{x^{\frac{1-n}{n}}}{n} \\
	&=\frac{x^{\frac{1-n}{n}}}{n}\left(\frac{(n+1)(n-j)}{j+1}\left(1 - \frac{n}{n+1}x^{-\frac{n}{n+1} -\frac{1-n}{n}}\right) - 1\right) \\
	&=\frac{x^{\frac{1-n}{n}}}{n}\left(\frac{(n+1)(n-j)}{j+1}\left(1 - \frac{n}{n+1}x^{-\frac{1}{n(n+1)}}\right) - 1\right).
\end{align*}
From here we see that there exists a single root,
\begin{align*}
	&\frac{x^{\frac{1-n}{n}}}{n}\left(\frac{(n+1)(n-j)}{j+1}\left(1 - \frac{n}{n+1}x^{-\frac{1}{n(n+1)}}\right) - 1\right) \overset{\text{set}}{=} 0, \\
	&\qquad\implies \frac{(n+1)(n-j)}{j+1}\left(1 - \frac{n}{n+1}x^{-\frac{1}{n(n+1)}}\right) = 1, \\
	&\qquad\implies \frac{n}{n+1}x^{-\frac{1}{n(n+1)}} = 1 - \frac{j+1}{(n+1)(n-j)}, \\
	&\qquad\implies x^* = \left(\frac{n+1}{n}\left(1 - \frac{j+1}{(n+1)(n-j)}\right)\right)^{-n(n+1)}.
\end{align*}
Simplifying further yields
\begin{align*}
	&\left(\frac{n+1}{n}\left(1 - \frac{j+1}{(n+1)(n-j)}\right)\right)^{-n(n+1)} 
	 = \left(\frac{n+1}{n} - \frac{j+1}{n(n-j)}\right)^{-n(n+1)} \\
	 &\qquad= \left(\frac{(n+1)(n-j)}{n(n-j)} - \frac{j+1}{n(n-j)}\right)^{-n(n+1)} \\
	 &\qquad= \left(\frac{n(n-j) + n - 2j+1}{n(n-j)} \right)^{-n(n+1)}.
\end{align*}
We have $x^* \geq 1$ whenever $n - 2j + 1 \leq 0$. We know that $f_{n,j}(0) = 1$ and $f_{n,j}(1) = 0$, and this implies that $f(x) > 0$ for all $x \in (0,1)$. Now consider when $n - 2j + 1 > 0$. We can rewrite \eqref{eq:fnj} as
$$
  f_{n,j}(x) = \frac{(n+1)(n-1) - (j+1)}{j+1}\xn - \frac{(n+1)(n-1)}{j+1}\xnone + 1.
$$
From here we can see that $f_{n,j}(x)$ is monotonic by noting that $n - 2j + 1 = (n-j)- (j+1)$. Therefore, when $\xn > j/n$ we can conclude that
$$
  I_{\xn}(n-j, j+1) - I_{\xnone}(n-j+1, n-j) \geq 0.
$$
