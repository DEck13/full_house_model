\subsection{Proof of Theorem 2.2}
\begin{proof}
Fix a population $i$ and suppress the index $i$. First, since $F_X$ is continuous, we can express well-defined order statistics as $X_{(1)} < \cdots < X_{(N)}$. By rank-based selection, the observed system consists of indices $j$ such that $r_j \in \mathcal{R}$, yielding observed pairs $(Y_{(j)}, X_{(r_j)})$, for $j = 1,\dots,n$. By assumption, for each observed rank $j$ there exists a strictly increasing measurable function $\psi_j$ such that
$$
  Y_{(j)} = \psi_j(X_{(r_j)}) \quad \text{a.s.}
$$
We now identify the distribution function $F_{Y_{(j)}}$ for fixed $j$. Since $\psi_j$ is strictly increasing, it admits a measurable inverse on its range. For any $y \in \mathbb{R}$,
$$
\Prob(Y_{(j)} \le y) 
	= \Prob(\psi_j(X_{(r_j)}) \le y)
    = \Prob(X_{(r_j)} \le \psi_j^{-1}(y)).
$$
Hence, $F_{Y_{(j)}}(y) = F_{X_{(r_j)}}(\psi_j^{-1}(y))$, which implies that
$$
  F_{Y_{(j)}}(\psi_j(x)) = F_{X_{(r_j)}}(x)
	\quad \text{for } F_{X_{(r_j)}}\text{-a.e. } x.
$$

We now show that $Y_{(j)}$ is recovered from the Full House Modeling construction. Since $F_X$ is continuous, the distribution of the order statistic $X_{(r_j)}$ is also continuous and strictly increasing on the interior of its support, hence admits a well-defined generalized inverse. Likewise, $F_{Y_{(j)}}$ is continuous and strictly increasing. Thus both admit well-defined inverses. Let $u = F_{X_{(r_j)}}(x)$. Then $F_{Y_{(j)}}(\psi_j(x)) = u$, so that 
$$
\psi_j(x)
	= F_{Y_{(j)}}^{-1}(u)
	= F_{Y_{(j)}}^{-1}\!\big(F_{X_{(r_j)}}(x)\big)
$$
for $F_{X_{(r_j)}}$-almost every $x$. Therefore,
$$
Y_{(j)} = F_{Y_{(j)}}^{-1} \circ F_{X_{(r_j)}}(X_{(r_j)})
	\quad \text{a.s.}	
$$
We now verify the uniqueness of the above. Let $U = F_{X_{(r_j)}}(X_{(r_j)}) \sim \mathrm{Unif}(0,1)$ by the probability integral transform, and define
$$
  g(u) := \psi_j\!\big(F_{X_{(r_j)}}^{-1}(u)\big).
$$
Then $g$ is strictly increasing and $g(U) \sim F_{Y_{(j)}}$. By the fact that if $U\sim\mathrm{Unif}(0,1)$ and $g$ is increasing, then the distribution of $g(U)$ uniquely determines $g$ almost everywhere, implying $g(u)=F^{-1}_{Y_{(j)}}(u)$ a.e. It follows that 
$$
\psi_j(x) = F_{Y_{(j)}}^{-1}\!\big(F_{X_{(r_j)}}(x)\big)
	\quad \text{for } F_{X_{(r_j)}}\text{-a.e. } x.
$$
Thus, the strictly increasing coupling is unique up to null sets. The representation holds for each observed rank $j$, and uniqueness implies that the joint distribution of $(X,Y)$ coincides with that induced by the Full House Model.
\end{proof}






%%%% Proof of Lemma 3.2
\subsection{Proof of Lemma~\ref{lemma:rep}}

\begin{proof}
Define $f_{1,j}(\cdot) = F_{Y_{1, (j)}}^{-1} \circ F_{X_{1,(r_{1,j})}} (\cdot)$ and 
$f_{2,k}(\cdot) = F_{Y_{2, (k)}}^{-1} \circ F_{X_{2,(r_{2,k})}} (\cdot)$. Then we have 
\begin{align*}
  &\Prob\left(X_{1,(r_{1,j})} > X_{2,(r_{2,k})}, Y_{1,(j)} < Y_{2,(k)}\right) \\
  &\qquad= \Prob\left(X_{1,(r_{1,j})} > X_{2,(r_{2,k})}, f_{1,j}(X_{1,(r_{1,j})}) < f_{2,k}(X_{2,(r_{2,k})})\right) \\
  &\qquad= \Prob\left( X_{2,(r_{2,k})} < X_{1,(r_{1,j})}  < h_{j,k}\left(X_{2,(r_{2,k})}\right)\right),
\end{align*}
where $h_{j,k}(\cdot) = (f_{1,j}^{-1} \circ f_{2,k})(\cdot)$. %Existence of $f_i$ follows directly from the existence of $F_Y^{-1}(y)$, and existence of $h_{i,j}$ follows directly from the existence of $f_i$ and the existence of $F_X^{-1}(x)$.
A similar technique can be applied to the second equation to obtain the result.
\end{proof}


%%%% Proof of Theorem 3.4
\subsection{Proof of Theorem~\ref{theorem:samedistgeneral}} 

\begin{proof}
By Lemma~\ref{lemma:rep},
$$
	\Prob\left(X_{1,(r_{1,j})} > X_{2,(r_{2,k})}, Y_{1,(j)} < Y_{2,(k)}\right) = \Prob\left( X_{2,(r_{2,k})} < X_{1,(r_{1,j})}  < h_{j,k}\left(X_{2,(r_{2,k})}\right)\right),
$$
where $h_{j,k}(\cdot) = F_{X_{1,(r_{1,j})}}^{-1} \circ F_{Y_{1, (j)}} \circ F_{Y_{2, (k)}}^{-1} \circ F_{X_{2,(r_{2,k})}} (\cdot)$. Under the assumption $F_{Y_1}=F_{Y_2}=F_Y$, the distributions of the order statistics admit Beta representations, which allow us to express the mapping $h_{j,k}$ in terms of regularized incomplete beta functions.
\begin{align*}
	&\Prob\left( X_{2,(r_{2,k})} < X_{1,(r_{1,j})}  < h_{j,k}\left(X_{2,(r_{2,k})}\right)\right) \\
	&= \int_{-\infty}^{+\infty}\left\{\int_{x}^{F_X^{-1} \circ I_{r_{1,j}, N_1-r_{1,j}+1}^{-1} \circ I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1} \circ I_{r_{2,k}, N_2-r_{2,k}+1} \circ F_X(x)} f_{X_{1,(r_{1,j})}}(y)\ dy\right\} \\ 
	&\qquad \times  f_{X_{2,(r_{2,k})}}(x) \ dx \\
	&= \int_{-\infty}^{+\infty}\left\{\int_{x}^{F_X^{-1} \circ I_{r_{1,j}, N_1-r_{1,j}+1}^{-1} \circ I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1} \circ I_{r_{2,k}, N_2-r_{2,k}+1} \circ F_X(x)} d \ I_{ r_{1,j}, N_1- r_{1,j}+1}(F_X(y))\right\} \\
	&\qquad \times f_{X_{2,(r_{2,k})}}(x) \ dx \\
	&= \int_{-\infty}^{+\infty}\left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1} \circ F_X(x) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ F_X(x)\right\}^+ \\ 
	&\qquad \times f_{X_{2,(r_{2,k})}}(x) \ dx \\
	&= \int_{-\infty}^{+\infty}\left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1} \circ F_X(x) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ F_X(x)\right\}^+ \\  
	&\qquad d \ I_{ r_{2,k}, N_2- r_{2,k}+1}(F_X(x)) \\
	&= \int_0^1 \left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1}(t) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1}^{-1}(t)\right\}^+	 \ dt.
\end{align*}
This representation holds without additional assumptions. The positive-part operator appears because the integrand may change sign. Under condition~\eqref{eq:technical-condition}, the integrand is nonnegative, hence
\begin{align*}
	&\int_0^1 \left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1}(t) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1}^{-1}(t)\right\}^+\ dt \\
	&= \int_0^1 \left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1}(t) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1}^{-1}(t)\right\}\ dt.
\end{align*}
Thus,
\begin{align*}	
&\int_0^1 \left\{I_{j, n_1-j+1} \circ I_{k, n_2-k+1}^{-1}(t) - I_{ r_{1,j}, N_1- r_{1,j}+1} \circ I_{ r_{2,k}, N_2- r_{2,k}+1}^{-1}(t)\right\}^+\ dt \\
	&\qquad= \int_0^1 I_{j, n_1 - j + 1}(x)\, dI_{k, n_2 - k + 1}(x)
- \int_0^1 I_{r_{1,j}, N_1 - r_{1,j} + 1}(x)\, dI_{r_{2,k}, N_2 - r_{2,k} + 1}(x)\\
&\qquad= \int_0^1 I_{j, n_1 - j + 1}(x)\,
\frac{x^{k-1}(1-x)^{n_2-k}}{B(k, n_2-k+1)} \, dx \\
&\qquad\qquad- \int_0^1 I_{r_{1,j}, N_1-r_{1,j}+1}(x)\,
\frac{x^{r_{2,k}-1}(1-x)^{N_2-r_{2,k}}}{B(r_{2,k}, N_2-r_{2,k}+1)} \, dx\\
&\qquad = \int_0^1 \sum_{p=j}^{n_1}\binom{n_1}{p}
x^p(1-x)^{n_1-p}
\frac{x^{k-1}(1-x)^{n_2-k}}{B(k, n_2-k+1)} \, dx \\
&\qquad\qquad - \int_0^1 \sum_{p=r_{1,j}}^{N_1}\binom{N_1}{p}
x^p(1-x)^{N_1-p}
\frac{x^{r_{2,k}-1}(1-x)^{N_2-r_{2,k}}}{B(r_{2,k}, N_2-r_{2,k}+1)} \, dx\\ 
&\qquad = \sum_{p=j}^{n_1}\binom{n_1}{p}
\frac{B(k+p, n_2+n_1-p-k+1)}{B(k, n_2-k+1)} \\
&\qquad\qquad - \sum_{p=r_{1,j}}^{N_1}\binom{N_1}{p}
\frac{B(r_{2,k}+p, N_2+N_1-p-r_{2,k}+1)}{B(r_{2,k}, N_2-r_{2,k}+1)}.
\end{align*}
\end{proof}



%%%% Proof of Proposition 3.5
\subsection{Proof of Proposition~\ref{prop:condition}}

\begin{proof}
Specification of the selection rule $S_{i,j} = 1 \;\Longleftrightarrow\; X_{i,j} \geq X_{i,(N_i - n_i + 1)}$ implies that $r_{i,j} = N_i - j$. When either $n_1 = N_1$ and $n_2 = N_2$ or $n_1 = n_2$ and $N_1 = N_2$ we have, for all $t \in (0,1)$, that
$$
  I_{N_2-j,j+1} \circ I_{N_1-j,j+1}^{-1} \circ I_{n_1-j,j+1} \circ I_{n_2-j,j+1}^{-1} (t) \geq t.
$$
The left hand side of the above equation increases whenever $n_1$ or $N_1$ are increased. This follows from the monotonicity of the inverse incomplete beta function in its parameters \citep{askitis2021logarithmic}, together with the fact that stochastic dominance is preserved under monotone transformations. Therefore,
$
  I_{N_2-j,j+1} \circ I_{N_1-j,j+1}^{-1} \circ I_{n_1-j,j+1} \circ I_{n_2-j,j+1}^{-1} (t) \geq t
$
whenever $\frac{n_1}{n_2}\leq 1$ and $\frac{N_2}{N_1} \leq 1$. Observe that
\begin{align*}
&I_{N_2-j,j+1} \circ I_{N_1-j,j+1}^{-1} \circ I_{n_1-j,j+1} \circ I_{n_2-j,j+1}^{-1} (t) \geq t, \\ 
&\quad\implies I_{n_1-j,j+1} \circ I_{n_2-j,j+1}^{-1} (t) \geq I_{N_1-j,j+1} \circ I_{N_2-j,j+1}^{-1}(t),   \\
&\quad\implies I_{n_1-j,j+1} \circ I_{n_2-j,j+1}^{-1} (x) - I_{N_1-j,j+1} \circ I_{N_2-j,j+1}^{-1}(t) \geq 0.
\end{align*}
Technical condition \eqref{eq:technical-condition} holds and the conclusion of Theorem~\ref{theorem:samedistgeneral} applies.
\end{proof}

%%%% Proof of Proposition 3.5
\subsection{Proof of Lemma~\ref{lemma:beta-quantile-density-comparison}}

\begin{proof}
The inverse-function derivative gives
\[
 q_s'(u)
 =\frac{B(s-j,j+1)}{q_s(u)^{s-j-1}(1-q_s(u))^j}
 =\frac{q_s(u)}{s h_s(u)}.
\]
Consequently,
\[
 h_s'(u)=1-\frac{j}{s(1-q_s(u))},
 \qquad
 h_s''(u)=-\frac{j q_s(u)}{s^2(1-q_s(u))^2h_s(u)}.
\]
Since $s-j>0$ and $j>0$, $h_s$ extends continuously to $[0,1]$
with $h_s(0)=h_s(1)=0$.

Put $m=h_n-h_N$. If $m$ were nonnegative at any point of $(0,1)$,
then it would attain a nonnegative maximum at some $u_0\in(0,1)$.
At this point,
\[
 m(u_0)\geq0,\qquad m'(u_0)=0,\qquad m''(u_0)\leq0.
\]
The equality $m'(u_0)=0$ implies
\[
 n(1-q_n(u_0))=N(1-q_N(u_0))=:c>0,
\]
and hence
\[
 q_N(u_0)-q_n(u_0)=c\left(\frac1n-\frac1N\right)>0.
\]
Because $h_n(u_0)\geq h_N(u_0)>0$, it follows that
\begin{align*}
 m''(u_0)
 &=\frac{j}{c^2}\left(
   \frac{q_N(u_0)}{h_N(u_0)}-
   \frac{q_n(u_0)}{h_n(u_0)}\right)\\
 &\geq\frac{j}{c^2}\,
   \frac{q_N(u_0)-q_n(u_0)}{h_n(u_0)}>0,
\end{align*}
a contradiction. Thus $m(u)<0$ for all $u\in(0,1)$.
\end{proof}

\subsection{Proof of Proposition~\ref{prop:condition2}}
\begin{proof}
Specification of the selection rule $S_{i,j} = 1 \;\Longleftrightarrow\; X_{i,j} \geq X_{i,(N_i - n_i + 1)}$ implies that $r_{i,j} = N_i - j$. Using the reduction of technical condition~\eqref{eq:technical-condition}
for this model, it suffices to show that
\[
 f(x):=I_{n_1-j,j+1}\bigl(I_{n_2-j,j+1}^{-1}(x)\bigr)
       -I_{N_1-j,j+1}\bigl(I_{N_2-j,j+1}^{-1}(x)\bigr)
 \geq0\qquad(0<x<1).
\]
If $j=0$, then $I_{a,1}(t)=t^a$ and
\[
 f(x)=x^{n_1/n_2}-x^{N_1/N_2}\geq0,
\]
because the ratio assumption gives $n_1/n_2\leq N_1/N_2$.
Henceforth assume $j>0$. In this case, the inequality is strict.

Let $d=n_2-n_1$ and $D=N_2-N_1$. The ratio assumption is equivalent to
\[
 \frac{d}{n_2}\geq\frac{D}{N_2}.
\]
Because $D>d\geq0$, this implies $d>0$ and $N_2>n_2$.
It also implies
$N_1\geq n_1N_2/n_2>n_1>j$, so all beta parameters are positive.

For $s\in\{n_2,N_2\}$, use the notation
\[
 q_s(x)=I_{s-j,j+1}^{-1}(x),
 \qquad
 h_s(x)=\frac{q_s(x)^{s-j}(1-q_s(x))^j}{sB(s-j,j+1)}.
\]
Define the positive constants
\[
 A=\frac{B(n_2-j,j+1)}{B(n_1-j,j+1)},
 \qquad
 C=\frac{B(N_2-j,j+1)}{B(N_1-j,j+1)}.
\]
Differentiating the two compositions yields
\[
 f'(x)=Aq_{n_2}(x)^{-d}-Cq_{N_2}(x)^{-D}
      =q_{n_2}(x)^{-d}\bigl(A-Cr(x)\bigr),
\]
where
\[
 r(x)=\frac{q_{n_2}(x)^d}{q_{N_2}(x)^D}.
\]
By Lemma~\ref{lemma:beta-quantile-density-comparison},
$h_{n_2}(x)<h_{N_2}(x)$. Therefore
\begin{align*}
 \frac{r'(x)}{r(x)}
 &=\frac{d}{n_2h_{n_2}(x)}-\frac{D}{N_2h_{N_2}(x)}\\
 &\geq\frac{d}{n_2}
    \left(\frac1{h_{n_2}(x)}-\frac1{h_{N_2}(x)}\right)>0.
\end{align*}
Thus $r$ is strictly increasing, and $A-Cr(x)$ is strictly decreasing.

The function $f$ extends continuously to $[0,1]$ with
$f(0)=f(1)=0$. By Rolle's theorem, there exists $x_*\in(0,1)$
such that $f'(x_*)=0$, equivalently $r(x_*)=A/C$.
Strict monotonicity of $r$ makes this point unique and gives
\[
 f'(x)>0\quad(0<x<x_*),
 \qquad
 f'(x)<0\quad(x_*<x<1).
\]
Together with $f(0)=f(1)=0$, these inequalities show that
$f(x)>0$ for every $x\in(0,1)$, which establishes the required
beta-function inequality.
\end{proof}



%%%% Proof of Proposition 4.2
\subsection{Proof of Proposition~\ref{prop:weakstodom}}

\begin{proof}
First, in combination with the Definition \ref{def:stodom} and the assumption $\frac{n_1}{N_1} \leq \frac{n_2}{N_2}$, we have 
$$
  F_{Y_1}(x)^{\frac{n_1}{N_1}} \geq F_{Y_2}(x)^{\frac{n_2}{N_2}} 
 \implies  
 x \geq F_{Y_1}^{-1}\left(F_{Y_2}(x)^{\frac{n_2N_1}{N_2n_1}}\right).
$$ 
By applying the Equation \ref{eq:pararep2} from the Lemma \ref{lemma:rep} we have 
\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
	= \Prob\left( g_{n_1,n_2}\left(Y_{2,(n_2)}\right) < Y_{1,(n_1)}  < Y_{2,(n_2)}\right) \\
	&\qquad = \Prob\left( F_{Y_1}^{-1}\left(F_{Y_2}(Y_{2,(n_2)})^{\frac{n_2N_1}{N_2n_1}}\right) < Y_{1,(n_1)}  < Y_{2,(n_2)}\right) \\
	&\qquad = \int_{-\infty}^{+\infty} \int_{F_{Y_1}^{-1}\left(F_{Y_2}(x)^{\frac{n_2N_1}{N_2n_1}}\right)}^{x} d F_{Y_1, (n_1)}(y) d F_{Y_2, (n_2)}(x) \\
	&\qquad = \int_{-\infty}^{+\infty} F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}}d F_{Y_2, (n_2)}(x)\\
	&\qquad = \int_0^1 F_{Y_1}\left( F_{Y_2}^{-1}(t^{\frac{1}{n_2}}) \right)^{n_1} - t^{\frac{N_1}{N_2}} dt. 
\end{align*}
Based on the Definition \ref{def:stodom}, we have
\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
	= \int_0^1 F_{Y_1}\left( F_{Y_2}^{-1}(t^{\frac{1}{n_2}}) \right)^{n_1} - t^{\frac{N_1}{N_2}} dt \\
	&\qquad \geq \int_0^1 F_{Y_2}\left( F_{Y_2}^{-1}(t^{\frac{1}{n_2}}) \right)^{n_1} - t^{\frac{N_1}{N_2}} dt \\
	&\qquad = \frac{n_2}{n_1+n_2} - \frac{N_2}{N_1+N_2},
\end{align*}
and
\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) 
	= \int_0^1 F_{Y_1}\left( F_{Y_2}^{-1}(t^{\frac{1}{n_2}}) \right)^{n_1} - t^{\frac{N_1}{N_2}} dt \\
	&\qquad \leq \int_0^1 \left(t^{1/n_2} + \varepsilon\right)^{n_1} - t^{\frac{N_1}{N_2}} dt\\
	&\qquad = n_2\sum_{i=0}^{n_1} \frac{{n_1 \choose i}\varepsilon^i}{n_1+n_2-i}- \frac{N_2}{N_1 + N_2}.
\end{align*}
\end{proof}



%%%% Proof of Theorem 4.3
\subsection{Proof of Theorem~\ref{theorem:normalgeneral}}

\begin{proof}
Equation \ref{eq:pararep2} from Lemma \ref{lemma:rep} yields
\begin{align*}
	g_{n_1, n_2}(x) = F_{Y_1}^{-1}\left(F_{Y_2}(x)^{\frac{n_2N_1}{N_2n_1}}\right) = \Phi^{-1}\left(\Phi\left(\frac{x-\mu_2}{\sigma_2}\right)^\frac{n_2N_1}{N_2n_1}\right)\sigma_1 +\mu_1.
\end{align*}
Let $A$ and $R$ be as defined, respectively, in \eqref{eq:A} and \eqref{eq:remainder}. We have,
\begin{align*}
  &\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)}\right) = \Prob\left( g_{n_1,n_2}\left(Y_{2,(n_2)}\right) < Y_{1,(n_1)}  < Y_{2,(n_2)}\right)\\
	&\qquad= \Prob\left( F_{Y_1}^{-1}\left(F_{Y_2}(Y_{2,(n_2)})^{\frac{n_2N_1}{N_2n_1}}\right) < Y_{1,(n_1)}  < Y_{2,(n_2)}\right)\\
	&\qquad= \int_A \int_{F_{Y_1}^{-1}\left(F_{Y_2}(x)^{\frac{n_2N_1}{N_2n_1}}\right)}^{x} d F_{Y_1, (n_1)}(y) d F_{Y_2, (n_2)}(x)\\	
	&\qquad= \int_{A}\left[ F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}}\right]d F_{Y_2, (n_2)}(x) \\
	&\qquad= \int_{-\infty}^{\infty} \left[F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}}\right]d F_{Y_2, (n_2)}(x) - \int_{A^c} \left[F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}}\right]d F_{Y_2, (n_2)}(x) \\
	&\qquad= \int_{-\infty}^{\infty}\left[ F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}}\right]d F_{Y_2, (n_2)}(x) + R.
\end{align*}
We continue with the left-hand side of the last equality,	
\begin{align*}
&\int_{-\infty}^{\infty} \left[ F_{Y_1}(x)^{n_1} - F_{Y_2}(x)^{\frac{n_2N_1}{N_2}} \right] d F_{Y_2,(n_2)}(x) 
= \int_0^1 \left[ F_{Y_1}\!\left( F_{Y_2}^{-1}(t^{1/n_2}) \right)^{n_1} - t^{N_1/N_2} \right] dt\\[6pt]
&\qquad= \int_{0}^{1} \left( \Phi\!\left(\frac{\Phi^{-1}(t^{1/n_2})\sigma_2 + \mu_2 - \mu_1}{\sigma_1}\right) \right)^{n_1} \, dt \;-\; \frac{N_2}{N_1 + N_2} \\[6pt]
&\qquad= \int_{-\infty}^{+\infty} 
\Phi\!\left(\frac{\sigma_2 x + \mu_2 - \mu_1}{\sigma_1}\right)^{n_1} 
\left( n_2 \Phi(x)^{n_2-1} \phi(x) \right)\, dx \;-\; \frac{N_2}{N_1 + N_2} \\[8pt]
&\qquad= n_2 \int_{-\infty}^{+\infty}
\Prob\!\left(
\begin{array}{l}
Z_{1,1} - \frac{\sigma_2}{\sigma_1}X \le \frac{\mu_2 - \mu_1}{\sigma_1},\, \dots,\,
Z_{1,n_1} - \frac{\sigma_2}{\sigma_1}X \le \frac{\mu_2 - \mu_1}{\sigma_1},\\[3pt]
Z_{2,1}-X \le 0,\, \dots,\, Z_{2,n_2-1}-X \le 0
\end{array}
\;\middle|\; X=x \right)
\phi(x)\, dx \\ 
&\qquad\qquad\;-\; \frac{N_2}{N_1 + N_2} \\[10pt]
&\qquad= n_2 \,
\Prob\!\left(
\begin{array}{l}
Z_{1,1} - \frac{\sigma_2}{\sigma_1}X \le \frac{\mu_2 - \mu_1}{\sigma_1},\, \dots,\,
Z_{1,n_1} - \frac{\sigma_2}{\sigma_1}X \le \frac{\mu_2 - \mu_1}{\sigma_1},\\[3pt]
Z_{2,1}-X \le 0,\, \dots,\, Z_{2,n_2-1}-X \le 0
\end{array}
\right)
\;-\; \frac{N_2}{N_1 + N_2} \\[10pt]
&\qquad= n_2\, F_{\mathbf{W}}\!\left( \mathbf{w}\right) \;-\; \frac{N_2}{N_1 + N_2},
\end{align*}
where the first $n_1$ elements of $\mathbf{w}$ is $\frac{\mu_2 - \mu_1}{\sigma_1}$ and the last $n_2-1$ elements are 0. The distribution function $F_{\mathbf{W}}$ corresponds to random variable $\mathbf{W} \sim \mathcal{N}_{(n_1 + n_2 - 1)}(\boldsymbol{0}, \boldsymbol{\Sigma})$ where $\boldsymbol{\Sigma}$ is as defined in \eqref{eq:Sigma} and \eqref{eq:Sigmastructure}.

The bound on the remainder \eqref{eq:remainder-bound} follows from the argument in Section~\ref{subsec:normal}.
\end{proof}

$$
F_{\mathbf W}(\mathbf w) = \Prob\left(Z_0 \le \frac{\mu_2 - \mu_1}{\sigma_2} - \frac{\sigma_1}{\sigma_2}\sqrt{2 \ln n_1}, \; \; Z_0 \le -\sqrt{2 \ln (n_2-1)}\right).

$$

\end{align*}
are satisfied.

$$
  \frac{\widehat F_{\mathbf W}(\mathbf w)}{F_{\mathbf W}(\mathbf w)}
	\xrightarrow{P}1.
$$

\begin{align*}

  &\qquad = n_2\left(\widehat F_{\mathbf W}(\mathbf w) - F_{\mathbf W}(\mathbf w)\right) + \left(\widehat R - R\right)
  \xrightarrow{P}0,	
\end{align*}
noting that $n_2\left(\widehat F_{\mathbf W}(\mathbf w) - F_{\mathbf W}(\mathbf w)\right) = n_2F_{\mathbf W}(\mathbf w)\left(\frac{\widehat F_{\mathbf W}(\mathbf w)}{F_{\mathbf W}(\mathbf w)} - 1\right) = o_P(1)$.
\end{proof}

%%%% Proof of Proposition 3.5
\subsection{Proof of Proposition~\ref{prop:common-copula-bounds}}

\begin{proof}
Let \((U_1,V_1)\) and \((U_2,V_2)\) be independent draws from \(C\), and
write
\[
A_b=\{U_1<U_2^b\},\qquad
B_a=\{V_1>V_2^a\},\qquad
D=A_1\cap B_1.
\]
The paradox event is \(E=A_b\cap B_a\). Because the two pairs are iid and
their margins are continuous, the two orientations of a discordant pair have
equal probability. Hence
\[
\Pr(D)=\frac12\Pr\{(U_1-U_2)(V_1-V_2)<0\}
=\frac{1-\tau}{4}=d_\tau.
\]
Uniformity of the margins also gives
\[
\Pr(A_b)=\int_0^1u^bdu=p_U,
\qquad
\Pr(B_a)=\int_0^1(1-v^a)dv=p_V.
\]

If \(b\leq1\), then \(A_1\subseteq A_b\) and
\(\Pr(A_b\setminus A_1)=e_U\); if \(b\geq1\), then
\(A_b\subseteq A_1\) and \(\Pr(A_1\setminus A_b)=c_U\). Similarly,
if \(a\geq1\), then \(B_1\subseteq B_a\) with added probability \(e_V\),
whereas if \(a\leq1\), then \(B_a\subseteq B_1\) with removed probability
\(c_V\). Therefore,
\[
D\setminus E
\subseteq (A_1\setminus A_b)\cup(B_1\setminus B_a),
\qquad
E\setminus D
\subseteq (A_b\setminus A_1)\cup(B_a\setminus B_1),
\]
which implies
\[
d_\tau-c_U-c_V\leq\Pr(E)\leq d_\tau+e_U+e_V.
\]
The Fr\'echet bounds for \(E=A_b\cap B_a\) give
\[
\max(0,p_U+p_V-1)\leq\Pr(E)\leq\min(p_U,p_V).
\]
Intersecting the two intervals proves the stated bounds. If \(a=b=1\), all
four adjustment terms vanish and \(E=D\), so both bounds equal \(d_\tau\).
If \(\tau=1\) and \(b\leq1\leq a\), then \(c_U=c_V=0\) and
\(e_U+e_V=p_U+p_V-1\), so the bounds again coincide.
\end{proof}

The four sensitivity panels (A)--(D) match those in the main text: (A) varies $(N_1,N_2)$ with $(n_1,n_2)$ fixed; (B) varies $(n_1,n_2)$ with $(N_1,N_2)$ fixed; (C) varies $(n_2,N_1)$ with $(n_1,N_2)$ fixed; (D) varies $(n_1,N_2)$ with $(n_2,N_1)$ fixed.

Figures~\ref{Fig:sensitivity_normal_allk}--\ref{Fig:sensitivity_gamma_allk} report the full set of surfaces for $k=0,1,2,3$ under $Y\sim\mathcal N(0,1)$ and $Y\sim\mathrm{Gamma}(\alpha = 2,\beta = 1)$, respectively. Across all $k$ and both achievement distributions, the qualitative dependence on $(n_1,n_2,N_1,N_2)$ remains stable: increasing $n_2$ relative to $n_1$ increases the probability of the observed-achievement reversal, while increasing $N_1$ relative to $N_2$ increases the probability of the latent-aptitude reversal. As a result, the paradox probability is consistently largest when these two effects reinforce each other (large $n_2/n_1$ and large $N_1/N_2$), and smallest when they oppose each other.

Varying $k$ primarily changes the overall magnitude of the paradox probability, but it does not alter the location of high-probability regimes in the $(n_1,n_2,N_1,N_2)$ design. Replacing $Y\sim\mathcal N(0,1)$ by the skewed Gamma distribution, $Y\sim\mathrm{Gamma}(\alpha = 2,\beta = 1)$, leads to similar surfaces in shape, indicating that the qualitative mechanisms are not driven by symmetry of the achievement distribution.

\subsection{Ratio of $N_2$ and $N_1$ Under a Skewed Gamma Achievement Distribution}\label{app:ratio_gamma}

We repeat the analysis of Section~\ref{sec:ratioeffect} (same $(n_1,n_2)$, $N_1$ grid, and $c$ grid) but replace the Normal achievement distribution by $Y\sim \mathrm{Gamma}(\alpha = 2,\beta = 1)$, which is right-skewed. The paradox probabilities are estimated using $10{,}000$ Monte Carlo repetitions for each configuration, and we report results for $k=0,1,2,3$ (largest through fourth-largest achiever). Figure~\ref{Fig:ratioeffect_gamma} shows that the dependence on $c$ is qualitatively the same as in the Normal case: the paradox probability decreases as $c$ increases, and conditional on $c$ it exhibits little sensitivity to $N_1$ over the range considered. Thus, the ratio effect described in the main text is robust to moderate skewness in the achievement distribution.
