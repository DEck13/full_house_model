\section{Monotonicity}

\begin{lemma}\label{lemma:monotonicity}
Let $1 \leq j < n$ and let $x \in (0,1)$. Let $I_{\alpha, \beta}(x)$ be a regularized incomplete beta function as in Definition~\ref{def:ribf}. Then, $I_{n-j, j+1}(\xn) \geq I_{n-j+1, j+1}(\xnone)$.
\end{lemma}
%
%\begin{center}
%	\textbf{Need to fix notation; should be $I_{\alpha,\beta}(x)$}
%\end{center}

\begin{proof}
We can define a function $f_{n,j}(x)$ as
\begin{align*}
  f_{n,j}(x) & = I_{n-j, j+1}(\xn) - I_{n-j+1, j+1}(\xnone) \\
  &= \int_0^{\xn}\frac{y^{n-j-1}(1-y)^j}{B(n-j,j+1)}dy 
    - \int_0^{\xnone}\frac{y^{n-j}(1-y)^j}{B(n-j+1,j+1)}dy.
\end{align*}
An application of the chain rule yields 
\begin{align*}
  &f_{n,j}^\prime(x) = \frac{\left(\xn\right)^{n-j-1}\left(1-\xn\right)^j\left(x^{\frac{1-n}{n}}\right)}{nB(n-j,j+1)} 
    - \frac{\left(\xnone\right)^{n-j}\left(1-\xnone\right)^j\left(x^{-\frac{n}{n+1}}\right)}{(n+1)B(n-j+1,j+1)} \\
  &= \frac{\left(\xn\right)^{n-j-1}\left(1-\xn\right)^j\left(x^{\frac{1-n}{n}}\right)}{nB(n-j,j+1)} 
    - \frac{\left(\xnone\right)^{n-j}\left(1-\xnone\right)^j\left(x^{-\frac{n}{n+1}}\right)}{(n-j)B(n-j,j+1)} \\    
  &= \frac{1}{B(n-j,j+1)}\left(\frac{\left(\xn\right)^{n-j-1}\left(1-\xn\right)^j\left(x^{\frac{1-n}{n}}\right)}{n} 
    - \frac{\left(\xnone\right)^{n-j}\left(1-\xnone\right)^j\left(x^{-\frac{n}{n+1}}\right)}{n-j}\right),
\end{align*}
where we used $B(n-j+1,j+1) = \frac{n-j}{n+1}B(n-j,j+1)$ in the second equality. Now define $A_{n,j} = \{x:f_{n,j}^\prime(x) \geq 0\}$. For $x \in A_{n,j}$ we have
\begin{align*}
	&\frac{\left(\xn\right)^{n-j-1}\left(1-\xn\right)^j\left(x^{\frac{1-n}{n}}\right)}{n} 
    \geq \frac{\left(\xnone\right)^{n-j}\left(1-\xnone\right)^j\left(x^{-\frac{n}{n+1}}\right)}{n-j} \\
    &\qquad\implies \frac{\left(\xn\right)^{n-j-1}\left(1-\xn\right)^j\left(x^{\frac{1-n}{n}}\right)}{\left(\xnone\right)^{n-j}\left(1-\xnone\right)^j\left(x^{-\frac{n}{n+1}}\right)} \geq \frac{n}{n-j} \\
    &\qquad\implies x^{-\frac{j}{n(n+1)}}\left(\frac{1-\xn}{1-\xnone}\right)^j \geq \frac{n}{n-j} \\
    &\qquad\implies \frac{x^{-\frac{1}{n(n+1)}}-\xnone}{1-\xnone} \geq \left(\frac{n}{n-j}\right)^\frac{1}{j}.
\end{align*}
Define $g_n(x) = \frac{x^{-\frac{1}{n(n+1)}}-\xnone}{1-\xnone}$. Then 
\begin{align*}
	g_n(x) = \frac{x^{-\frac{1}{n(n+1)}}-\xnone}{1-\xnone} = \frac{x^{-\frac{1}{n(n+1)}} - 1}{1-\xnone} + 1,
\end{align*}
and
\begin{align*}
	g^{'}_n(x) &= \frac{1}{(1-\xnone)^2}\left(-\frac{1}{n(n+1)}x^{-\frac{1}{n(n+1)}-1}(1-\xnone) - (x^{-\frac{1}{n(n+1)}} - 1)(-\frac{1}{n+1}x^{\frac{1}{n+1}-1})\right)\\
	&= \frac{1}{(1-\xnone)^2}\left( \frac{1}{n}x^{-\frac{n^2+1}{n(n+1)}} - \frac{1}{n+1}x^{-\frac{n}{n+1}} - \frac{1}{n(n+1)}x^{-\frac{n^2 + n + 1}{n(n+1)}}\right)\\
	&= \frac{x^{-\frac{n^2+1}{n(n+1)}}}{n(n+1)(1-\xnone)^2}\left((n+1) - nx^{\frac{1}{n(n+1)}} - x^{-\frac{1}{n+1}}\right).
\end{align*}
Now define $l_n(x) = (n+1) - nx^{\frac{1}{n(n+1)}} - x^{-\frac{1}{n+1}}$ and see that
\begin{align*}
	l^{'}_n(x) = -\frac{1}{n+1}x^{\frac{1}{n(n+1)} - 1} + \frac{1}{n+1}x^{-\frac{1}{n+1} - 1} = \frac{1}{n+1}x^{-\frac{n+2}{n+1}}\left( 1 - \xn \right).
\end{align*}
Thus, $l_n(x)$ is an increasing function for $x \in (0,1)$ and $\sup_{x \in (0,1)}l_n(x) = 0$. Therefore, $l_n(x) < 0$ for $x \in (0,1)$. Thus, $g_n(x)$ is an decreasing function, and $g_n(x)\to1$ as $x\to1$. We can see that $g_n(x) \in (1,+\infty)$ for $x \in (0,1)$. Thus $A_{n,j} = (0,x^*_{n,j}]$ where $x^*_{n,j}$ is such that $g_n(x^*_{n,j}) = (\frac{n}{n-j})^j$. Putting it all together, $f_{n,j}(0) = f_{n,j}(1) = 0$ and this implies that $I_{n-j, j+1}(\xn) \geq I_{n-j+1, j+1}(\xnone)$ for all $x \in (0,1)$.
\end{proof}

\begin{equation} \label{eq:seriescondi}
\begin{split}

  &\Phi(W_{n-1} = w_{n-1} | W_1 = w_1, W_2 = w_2, \dots, W_{n-2} = w_{n-2}) \times \dots \times \\
  &\Phi(W_2 = w_2| W_1 = w_1) \times \Phi(W_1 = w_1),
\end{split}
\end{equation}

where $\mathbf{W} = (W_1, W_2, \dots, W_n)$, $\mathbf{w} = (w_1, w_2, \dots, w_n)$.

%\begin{align*}

Then in general, for any $k \in \{2, \dots, n_1+n_2-1\}$, the conditional distribution of $W_k$ given $W_1, \dots, W_{k-1}$ is 

$$W_k \mid W_1=w_1, \ldots, W_{k-1}=w_{k-1} \sim \mathcal{N}\left(\mu_k, \sigma_k^2\right)$$ 

with 

$$\mu_k=\boldsymbol{\Sigma}_{k, 1:(k-1)} \boldsymbol{\Sigma}_{1:(k-1), 1:(k-1)}^{-1} \mathbf{w}_{1:(k-1)}$$

$$\sigma_k^2=\boldsymbol{\Sigma}_{k, k}-\boldsymbol{\Sigma}_{k, 1:(k-1)} \boldsymbol{\Sigma}_{1:(k-1), 1:(k-1)}^{-1} \boldsymbol{\Sigma}_{1:(k-1), k}$$

Here:
$\boldsymbol{\Sigma}_{k, 1:(k-1)}$ is a $1 \times(k-1)$ row vector: the covariances between $W_k$ and $W_1, \ldots, W_{k-1}$
$\boldsymbol{\Sigma}_{1:(k-1), 1:(k-1)}$ is a $(k-1) \times(k-1)$ covariance matrix, 
$\boldsymbol{\Sigma}_{k, k}$ is the $k$th element on the diagonal of the covariance matrix $\boldsymbol{\Sigma}$, and $\mathbf{w}_{1:(k-1)}=\left(w_1, \ldots, w_{k-1}\right)^T$.

Then we will show the the parameters $\mu$ and $\sigma^2$ of the conditional univariate normal distribution in the set of Equation (\ref{eq:wn}) are finite.

\section{Recursion}

\begin{prop} \label{prop:recursion}
Suppose that conditions of Theorem~\ref{theorem:samedistgeneral} where the system inclusion function is $1_i(X_{i,j} \geq X_{(N_i - n_i + 1)})$, $i = 1,2$. Suppose that $n_2 \geq n_1 > 1$, $N_1 \geq N_2 > 1$, and fix $k=j$. Then condition~\eqref{eq:technical-condition} holds, and the conclusions of Theorem~\ref{theorem:samedistgeneral} follow.
\end{prop}

\begin{proof}
For ease of notation we will let $g_{n,i}(x) = I_{i, n-i+1}(x)$.
%Supposed
%$$f(x) = g_{n_1, k} \circ g_{n_2, j}^{-1}(x) - g_{N_1, r_{1,k}} \circ g_{N_2, r_{2,j}}^{-1}(x)$$
%Given the property of the system inclusion mechanism, $g(X_{i,j},\textbf{X}_{i,-j}) = I(X_{i,j} \geq X_{N_i-n_i+1})$, and the condtion $k=j$, we have
%$$f(x) = g_{n_1, n_1-j} \circ g_{n_2, n_2-j}^{-1}(x) - g_{N_1, N_1 - j} \circ g_{N_2, N_2-j}^{-1}(x)$$
The proof will use two induction arguments. For the base case let $n_1 = n_2$ and $N_1 = N_2$. With these specifications we have that 
$$
  g_{n_1, n_1-j} \circ g_{n_2, n_2-j}^{-1}(x) - g_{N_1, N_1 - j} \circ g_{N_2, N_2-j}^{-1}(x) = 0.
$$
For the induction step, choose some $m \leq n_1 - 1$ and suppose that 
\begin{equation} \label{eq:inductstep}
  g_{n_1 -m, n_1-m -j} \circ g_{n_2, n_2-j}^{-1}(t) - g_{N_1, N_1 - j} \circ g_{N_2, N_2 - j}^{-1}(t) \geq 0,	
\end{equation}
for $t \in (0,1)$ where $n_2 \geq n_1$ and $N_1 \geq N_2$. An integration by parts calculation applied to regularized incomplete beta functions gives
$$
  I_{\alpha,\beta}(x) = I_{\alpha+1,\beta}(x) + \frac{x^\alpha(1-x)^\beta}{\alpha B(\alpha,\beta)}.
$$
Applying the above to \eqref{eq:inductstep} yields
\begin{align*}
	&g_{n_1-m-1, n_1-m-1-j} \circ g_{n_2, n_2-j}^{-1}(t) - g_{N_1, N_1 - j} \circ g_{N_2, N_2 - j}^{-1}(t) \\ 
	&\qquad= g_{n_1-m, n_1-m-j} \circ g_{n_2, n_2-j}^{-1}(t) - g_{N_1, N_1 -j} \circ g_{N_2, N_2-j}^{-1}(t) \\
	&\qquad\qquad+ \frac{\left[g_{n_2, n_2-j}^{-1}(t)\right]^{n_1-m-1-j}\left[1 - g_{n_2, n_2-j}^{-1}(t)\right]^{j+1}}{(n_1-m-1-j)B(n_1-m-1-j,j+1)} \\
	&\qquad\geq 0.
\end{align*}
A similar argument holds for iterative increases to $N_1$. Thus, condition~\eqref{eq:technical-condition} holds under are assumed conditions and the conclusions of Theorem~\ref{theorem:samedistgeneral} follow.
\end{proof}

\frac{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & \hdots &\hdots & \frac{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2}\\
\end{bmatrix}\\
	&= \begin{bmatrix}
1-\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & -\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & \hdots & -\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} \\
-\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & \ddots & \ddots  & \vdots \\
\vdots & \ddots & \ddots & -\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} \\
-\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & \dots & -\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2} & 1-\frac{1}{(\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1)n_2}\\
\end{bmatrix}_{(n_2-1) \times (n_2-1)}
\end{align*}

Then we have 

\begin{align*}
	\Phi(\mathbf{w}, \Sigma) \leq c^{n_1+n_2-1}
\end{align*}

Therefore, 

\begin{align*}
	 \Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_1} \leq \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_1} c^{n_1+n_2-1}
\end{align*}

Based on the structure of the $\Sigma^{-1}$ and $\frac{\partial \Sigma}{\partial \sigma_1}$, we have

\begin{align*}
	&\qquad \operatorname{tr}(\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_1})\\
	&= \operatorname{tr}\left(\begin{bmatrix}
	B_{11} & B_{12}\\
	B_{21} & B_{22}\\
\end{bmatrix} \begin{bmatrix}
	A_{11} & A_{12}\\
	A_{21} & A_{22}\\
\end{bmatrix}\right)\\
&= \operatorname{tr}(B_{11}A_{11} + B_{12}A_{21}) + tr(B_{21}A_{12} + B_{22}A_{22})\\
	&= n_1\left(\frac{-\frac{2\sigma_2^2}{\sigma_1^3}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} ( \frac{\sigma_2^2}{\sigma_1^2}\frac{n_1-1}{n_2}+1 - (n_1-1)\frac{\sigma_2^2}{\sigma_1^2}\frac{1}{n_2}) + \frac{\frac{\sigma_2}{\sigma_1}\frac{1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} \frac{\sigma_2}{\sigma_1^2} (n_2-1) \right)\\
	&+ (n_2-1)\left( n_1 \frac{\frac{\sigma_2}{\sigma_1}\frac{1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} \frac{\sigma_2}{\sigma_1^2}  \right)\\
	&= -2\frac{\frac{\sigma_2^2}{\sigma_1^3}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1}
\end{align*}

Now we have

\begin{align*}
	\operatorname{tr}(\Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_1} ) \leq 2\frac{\frac{\sigma_2^2}{\sigma_1^3}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} c^{n_1+n_2-1}
\end{align*}

Now we compute the term $\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_1}\right)$. Let rewrite the $\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x}$ into matrix form:

\begin{align*}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} &= \begin{bmatrix}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_1^2 d\mathbf{x} & \hdots & \int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_1x_n d\mathbf{x}\\
	\vdots & \ddots & \vdots \\
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_nx_1 d\mathbf{x} & \hdots &\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_n^2 d\mathbf{x}\\
\end{bmatrix}
\end{align*}

Define $\mathbf{w}_{-i}$ as the vector $\mathbf{w}$ with the $i$th component removed, and $\mathbf{w}_{-(i,j)}$ as the vector $\mathbf{w}$ with the $i$th and $j$ th components removed. Similarly, $\mathbf{x}_{-i}$ and $\mathbf{x}_{-(i,j)}$ are defined in the same manner. 

Then

\begin{align*}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_i^2 d\mathbf{x} &= \int_{-\infty}^{w_i} \left( \int_{-\infty}^{\mathbf{w}_{-i}} \phi(\mathbf{x}_{-i} | x_i) d\mathbf{x}_{-i} \right) \ x_i^2 \ \phi(x_i) dx_i\\
%	&\leq w_{n_1+n_2-2,max}^{n_1+n_2-2} \cdot Var(X_i)\\
%    &\leq w_{n_1+n_2-2,max}^{n_1+n_2-3} \cdot Var(X_i)\\
    &\leq c^{n_1+n_2-3} \cdot Var(X_i)\\
\end{align*}

\begin{align*}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_ix_j d\mathbf{x} &= \int_{-\infty}^{w_i} \int_{-\infty}^{w_j} \left( \int_{-\infty}^{\mathbf{w}_{-(i,j)}} \phi(\mathbf{x}_{-(i,j)} | x_i, x_j) d\mathbf{x}_{-(i,j)} \right) \ x_ix_j \ \phi(x_i, x_j) dx_i dx_j\\
%	&\leq c^{n_1+n_2-3} \cdot Cov(X_i, X_j)\\
	&\leq c^{n_1+n_2-3} \cdot Cov(X_i, X_j)\\
\end{align*}

Replace $Var(X_i)$ and $Cov(X_i, X_j)$ using the elements of $\Sigma$, we have

\begin{align*}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} &= \begin{bmatrix}
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_1^2 d\mathbf{x} & \hdots & \int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_1x_n d\mathbf{x}\\
	\vdots & \ddots & \vdots \\
	\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_nx_1 d\mathbf{x} & \hdots &\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot x_n^2 d\mathbf{x}\\
\end{bmatrix}\\
&\leq \begin{bmatrix}
	S_{11} & S_{12}\\
	S_{21} & S_{22}\\
\end{bmatrix}
\end{align*}

where 

$$S_{11} = \begin{bmatrix}
c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}+1) & c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) & \hdots & c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) \\
c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) & \ddots & \ddots  & \vdots \\
\vdots & \ddots & \ddots & c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) \\
c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) & \dots & c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}) & c^{n_1+n_2-3}(\frac{\sigma_2^2}{\sigma_1^2}+1)\\
\end{bmatrix}_{n_1 \times n_1}$$
$$S_{12}^{T} = S_{21} = \begin{bmatrix}
c^{n_1+n_2-3}(\frac{\sigma_2}{\sigma_1}) & \hdots & \hdots & c^{n_1+n_2-3}(\frac{\sigma_2}{\sigma_1})\\
\vdots & \ddots &  & \vdots\\
\vdots &  & \ddots & \vdots\\
c^{n_1+n_2-3}(\frac{\sigma_2}{\sigma_1}) & \hdots &\hdots & c^{n_1+n_2-3}(\frac{\sigma_2}{\sigma_1})\\
\end{bmatrix}_{(n_2-1) \times n_1}$$
$$
S_{22} = \begin{bmatrix}
2c^{n_1+n_2-3} & c^{n_1+n_2-3} & \hdots & c^{n_1+n_2-3}\\
c^{n_1+n_2-3} & \ddots & \ddots & \vdots\\
\vdots & \ddots & \ddots & c^{n_1+n_2-3}\\
c^{n_1+n_2-3} & \hdots & c^{n_1+n_2-3} & 2c^{n_1+n_2-3}\\
\end{bmatrix}_{(n_2-1) \times (n_2-1)}$$

Then, $$\begin{bmatrix}
	S_{11} & S_{12}\\
	S_{21} & S_{22}\\
\end{bmatrix} = c^{n_1+n_2-3}*\Sigma$$

Therefore, followed by the results before, 

\begin{align*}
&\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_1}\right)\\
	&\leq \operatorname{tr}\left(\Sigma^{-1}\begin{bmatrix}
	S_{11} & S_{12}\\
	S_{21} & S_{22}\\
\end{bmatrix}\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_1} \right)\\
&=\operatorname{tr}\left(\Sigma^{-1}c^{n_1+n_2-3}*\Sigma*\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_1} \right)\\
&= c^{n_1+n_2-3}\operatorname{tr}\left(\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_1} \right)\\
&= 2\frac{\frac{\sigma_2^2}{\sigma_1^3}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1}c^{n_1+n_2-3}
\end{align*}

Then 

\begin{align*}
	\operatorname{tr}(\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial \Sigma} \cdot \frac{\partial \Sigma}{\partial \sigma_1})
	&= \frac{1}{2}\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_1}\right) - \frac{1}{2}\operatorname{tr} \left( \Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_1} \right)\\
	&=  O_p(c^{n_1+n_2-3})
\end{align*}

By applying the similar method used for $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\mu_1}$, we have

\begin{align*}
	\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_1} 
	&= -\frac{(\mu_2 - \mu_1)n_1}{\sigma_1^2}\phi(w_1) \cdot \Phi(\tilde{\mathbf{w}}, \tilde{\Sigma}) + \operatorname{tr}(\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial \Sigma} \cdot \frac{\partial \Sigma}{\partial \sigma_1})\\
	&= -\frac{(\mu_2 - \mu_1)n_1}{\sigma_1^2}\phi(w_1) \cdot \Phi(\tilde{\mathbf{w}}, \tilde{\Sigma}) + O_p(c^{n_1+n_2-3}) \\
	&= O_P(n_1c^{n_1+n_2-3})
\end{align*}

Similarly, we compute $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2}$,

\begin{align*}
	\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2} &= \operatorname{tr}(\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial \Sigma} \cdot \frac{\partial \Sigma}{\partial \sigma_2}) \\
	&= \frac{1}{2}\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_2}\right) - \frac{1}{2}\operatorname{tr} \left( \Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_2} \right)\\
\end{align*}

To compute $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2}$, firstly we have

\begin{align*}
	\frac{\partial \Sigma}{\partial \sigma_2} &= \begin{bmatrix}
	\frac{\partial\Sigma_{11}}{\partial \sigma_1} & \frac{\partial\Sigma_{12}}{\partial \sigma_1}\\
	\frac{\partial\Sigma_{21}}{\partial \sigma_1} & \frac{\partial\Sigma_{22}}{\partial \sigma_1}\\
\end{bmatrix}\\
&= \begin{bmatrix}
	C_{11} & C_{12}\\
	C_{21} & C_{22}\\
\end{bmatrix}
\end{align*}

$$C_{11} = \begin{bmatrix}
\frac{2\sigma_2}{\sigma_1^2}
\end{bmatrix}_{n_1 \times n_1}, 
C_{12}^{T} = C_{21} = \begin{bmatrix}
\frac{1}{\sigma_1}
\end{bmatrix}_{(n_2-1) \times n_1}, C_{22} = \begin{bmatrix}
0\end{bmatrix}_{(n_2-1) \times (n_2-1)}$$

Then by applying Lemma \ref{lemma:wmax}, we have 

\begin{align*}
	 \Phi(\mathbf{w}, \Sigma) \leq c^{n_1+n_2-1}
\end{align*}

Then 

\begin{align*}
	 \Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_2} \leq \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_2} c^{n_1+n_2-1}
\end{align*}

Based on the structure of the $\Sigma^{-1}$ and $\frac{\partial \Sigma}{\partial \sigma_2}$, we have

\begin{align*}
	&\qquad \operatorname{tr}(\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_2})\\
	&= \operatorname{tr}\left(\begin{bmatrix}
	B_{11} & B_{12}\\
	B_{21} & B_{22}\\
\end{bmatrix} \begin{bmatrix}
	C_{11} & C_{12}\\
	C_{21} & C_{22}\\
\end{bmatrix}\right)\\
&= \operatorname{tr}(B_{11}C_{11} + B_{12}C_{21}) + tr(B_{21}C_{12} + B_{22}C_{22})\\
	&= n_1\left(\frac{\frac{2\sigma_2}{\sigma_1^2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} ( \frac{\sigma_2^2}{\sigma_1^2}\frac{n_1-1}{n_2}+1 - (n_1-1)\frac{\sigma_2^2}{\sigma_1^2}\frac{1}{n_2})\right) - \frac{\frac{\sigma_2}{\sigma_1}\frac{1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} \frac{1}{\sigma_1} n_1(n_2-1) \\
	&- (n_2-1)\left( n_1 \frac{\frac{\sigma_2}{\sigma_1}\frac{1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} \frac{1}{\sigma_1}  \right)\\
	&= \frac{\frac{2\sigma_2}{\sigma_1^2}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1}
\end{align*}

Now we have

\begin{align*}
	\operatorname{tr}(\Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_2} ) \leq \frac{\frac{2\sigma_2}{\sigma_1^2}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1} c^{n_1+n_2-1}
\end{align*}

\begin{align*}
	&\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_2}\right) \leq \operatorname{tr}\left(\Sigma^{-1}\begin{bmatrix}
	S_{11} & S_{12}\\
	S_{21} & S_{22}\\
\end{bmatrix}\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_2} \right)\\
&= c^{n_1+n_2-3}\operatorname{tr}\left(\Sigma^{-1}\frac{\partial \Sigma}{\partial \sigma_2}\right)\\
&= \frac{\frac{2\sigma_2}{\sigma_1^2}\frac{n_1}{n_2}}{\frac{\sigma_2^2}{\sigma_1^2}\frac{n_1}{n_2}+1}c^{n_1+n_2-3}
\end{align*}

\begin{align*}
	\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2} &= \operatorname{tr}(\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial \Sigma} \cdot \frac{\partial \Sigma}{\partial \sigma_2}) \\
	&= \frac{1}{2}\operatorname{tr}\left(\Sigma^{-1}\int_{-\infty}^\mathbf{w} \phi(\mathbf{x}) \cdot \mathbf{x} \mathbf{x}^t d\mathbf{x} \Sigma^{-1} \frac{\partial \Sigma}{\partial \sigma_2}\right) - \frac{1}{2}\operatorname{tr} \left( \Sigma^{-1} \Phi(\mathbf{w}, \Sigma)  \frac{\partial \Sigma}{\partial \sigma_2} \right)\\
	&= O_p(c^{n_1+n_2-3})
\end{align*}

Then we put $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\mu_1}$, $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\mu_2}$, $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_1}$, and $\frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2}$ together and based on the definition of root sample-size consistent estimator, 
$$
|\hat{\mu}_1 - \mu_1 | = O_p(n_1^{-1/2}), 
$$

$$
|\hat{\mu}_2 - \mu_2 | = O_p(n_2^{-1/2}), 
$$


$$
|\hat{\sigma}_1 - \sigma_1 | = O_p(n_1^{-1/2}), 
$$

$$
|\hat{\sigma}_2 - \sigma_2 | = O_p(n_2^{-1/2}), 
$$

we have

\begin{align*}
	\Phi(\mathbf{\hat{w}}, \hat{\Sigma}) - \Phi(\mathbf{w}, \Sigma) &\approx \frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\mu_1}(\hat{\mu_1} - \mu_1) + \frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\mu_2}(\hat{\mu_2} - \mu_2) + \frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_1}(\hat{\sigma_1} - \sigma_1) \\
	&+ \frac{\partial\Phi(\mathbf{w}, \Sigma)}{\partial\sigma_2}(\hat{\sigma_2} - \sigma_2) \\
	&= O_P(n_1^{\frac{1}{2}}c^{n_1+n_2-3})
	%&=  OP(n_1^{\frac{1}{2}}w_{max}^{n_1+n_2-2} + n_1n_2^{-\frac{1}{2}}w_{max}^{n_1+n_2-2} + n_1^{\frac{3}{2}}w_{max}^{n_1+n_2-3} + n_1^{\frac{1}{2}}n_2w_{max}^{n_1+n_2-3} \\
	%&+ n_1^{2}n_2^{-\frac{1}{2}}w_{max}^{n_1+n_2-3} + n_1n_2^{\frac{1}{2}}w_{max}^{n_1+n_2-3})
\end{align*}

Finally, 

\begin{align*}
	&\Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)} | \hat{\mu_1}. \hat{\mu_2}, \hat{\sigma_1}, \hat{\sigma_2} \right) - \Prob\left(X_{1,(N_1)} > X_{2,(N_2)}, Y_{1,(n_1)} < Y_{2,(n_2)} | \mu_1. \mu_2, \sigma_1, \sigma_2\right)\\
	& = O_P(n_1^{\frac{1}{2}}n_2c^{n_1+n_2-3} + n_1n_2^{\frac{1}{2}}c^{n_1+n_2-3})
	% \xrightarrow{n_1, n_2 \to \infty} 0
\end{align*}

\end{proof}

\section{Simulation Results for Theorem \ref{theorem:normalgeneral}}

We now show the simulation results of our Normal Distribution example under different configurations of $(\mu_1,\mu_2,\sigma_1,\sigma_2, n_1,n_2,N_1,N_2)$. We verify Theorem~\ref{theorem:normalgeneral} in the numerical simulation. Directly paradox probabilities from these theorems are compared with Monte Carlo paradox probabilities in a variety of settings. Each of these settings will involve comparisons of the observed statistics and latent aptitudes of the highest achievers, the pair $(Y_{i,(n_i)}, X_{i,(N_i)})$ across two populations.

The simulations will proceed as follows: Distributions for $Y_1$ and $Y_2$ are specified along with sample sizes $n_i$, $N_i$, $i = 1,2$. We will specify a Monte Carlo sample size of 2000. $
